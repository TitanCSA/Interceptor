cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(InterceptorUserMode VERSION 1.4.21.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Interceptor user-mode tools require Windows x64.")
endif()
if(NOT MSVC)
    message(FATAL_ERROR "Use the Visual Studio 2022 C++ toolchain.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8)
add_link_options("/PDBALTPATH:%_PDB%")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin/Native")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin/Native")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin/SDK")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/bin/Native")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/bin/Native")
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/bin/SDK")
endforeach()
add_library(InterceptorSDK STATIC Interceptor/Interceptor.cpp)
add_library(Interceptor::SDK ALIAS InterceptorSDK)
target_compile_features(InterceptorSDK PUBLIC cxx_std_17)
target_compile_definitions(InterceptorSDK PUBLIC UNICODE _UNICODE NOMINMAX)
target_include_directories(InterceptorSDK PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Interceptor")
target_link_libraries(InterceptorSDK PUBLIC setupapi newdev cfgmgr32 hid advapi32 user32 comctl32 ole32)

set(_interceptor_clients_default OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(_interceptor_clients_default ON)
endif()
option(INTERCEPTOR_BUILD_CLIENTS "Build the examples and installer native components" ${_interceptor_clients_default})
if(NOT INTERCEPTOR_BUILD_CLIENTS)
    return()
endif()
file(GLOB example_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/ExamplesConsole/*.cpp")
foreach(source IN LISTS example_sources)
    get_filename_component(name "${source}" NAME_WE)
    add_executable(${name} "ExamplesConsole/${name}.cpp")
    target_link_libraries(${name} PRIVATE Interceptor::SDK)
    target_link_options(${name} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()
add_executable(InterceptorExample InstallerUI/Native/Main.cpp InstallerUI/Native/Setup.cpp)
target_link_libraries(InterceptorExample PRIVATE Interceptor::SDK)
add_library(InterceptorViewerBridge SHARED ExamplesUI/ViewerBridge.cpp ExamplesUI/ViewerDecoder.cpp)
target_compile_definitions(InterceptorViewerBridge PRIVATE VIEWER_BRIDGE_EXPORTS)
target_link_libraries(InterceptorViewerBridge PRIVATE Interceptor::SDK)
add_executable(InterceptorDiagnosticWorker InstallerUI/Native/DiagnosticWorker.cpp)
target_compile_definitions(InterceptorExample PRIVATE IC_SDK_INTERNAL)
target_compile_definitions(InterceptorViewerBridge PRIVATE IC_SDK_INTERNAL)
target_compile_definitions(InterceptorDiagnosticWorker PRIVATE IC_SDK_INTERNAL)
target_link_libraries(InterceptorDiagnosticWorker PRIVATE Interceptor::SDK)
