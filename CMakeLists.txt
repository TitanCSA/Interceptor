cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(InterceptorUserMode VERSION 1.4.15.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Interceptor user-mode tools require Windows x64.")
endif()
if(NOT MSVC)
    message(FATAL_ERROR "Use the Visual Studio 2022 C++ toolchain.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8)
add_link_options("/PDBALTPATH:%_PDB%")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin/Native")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin/Native")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin/SDK")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/bin/Native")
    set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/bin/Native")
    set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config} "${CMAKE_SOURCE_DIR}/bin/SDK")
endforeach()
add_subdirectory(sdk)
set(examples 01_Echange_H_J 02_Ctrl_H_Hello 03_Inverse_Souris_Y
    04_Inverse_X_J_Boutons_HK 05_Clavier_Bloque_Affichage
    06_Souris_Bloquee_Affichage 07_Manette_HID_Inverse_X_Y 08_Clavier_Minimal)
foreach(name IN LISTS examples)
    add_executable(${name} "examples/console/${name}.cpp" examples/console/common/Runner.cpp)
    target_link_libraries(${name} PRIVATE Interceptor::SDK)
    target_link_options(${name} PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
endforeach()
add_executable(InterceptorExample installer/helper/Main.cpp installer/helper/Setup.cpp)
target_link_libraries(InterceptorExample PRIVATE Interceptor::SDK)
add_library(InterceptorViewerBridge SHARED examples/viewer/native/ViewerBridge.cpp
    examples/viewer/native/ViewerDecoder.cpp)
target_compile_definitions(InterceptorViewerBridge PRIVATE VIEWER_BRIDGE_EXPORTS)
target_link_libraries(InterceptorViewerBridge PRIVATE Interceptor::SDK)
add_executable(InterceptorDiagnosticWorker installer/ui/native/DiagnosticWorker.cpp)
target_link_libraries(InterceptorDiagnosticWorker PRIVATE Interceptor::SDK)
add_library(RecoveryBinding SHARED installer/recovery/RecoveryBinding.cpp)
target_link_libraries(RecoveryBinding PRIVATE setupapi newdev cfgmgr32 advapi32)
