cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(InterceptorSDK LANGUAGES CXX)
    set(_interceptor_example_default ON)
else()
    set(_interceptor_example_default OFF)
endif()
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Interceptor SDK requires a Windows x64 application.")
endif()
add_library(InterceptorSDK STATIC Interceptor.cpp InitDevices.cpp)
add_library(Interceptor::SDK ALIAS InterceptorSDK)
target_compile_features(InterceptorSDK PUBLIC cxx_std_17)
target_compile_definitions(InterceptorSDK PUBLIC UNICODE _UNICODE NOMINMAX)
target_include_directories(InterceptorSDK PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(InterceptorSDK PUBLIC setupapi newdev cfgmgr32 hid advapi32 user32 comctl32 ole32)
option(INTERCEPTOR_BUILD_MINIMAL_EXAMPLE "Build the minimal keyboard example" ${_interceptor_example_default})
if(INTERCEPTOR_BUILD_MINIMAL_EXAMPLE)
    add_executable(InterceptorMinimal "${CMAKE_CURRENT_SOURCE_DIR}/../examples/console/08_Clavier_Minimal.cpp")
    target_link_libraries(InterceptorMinimal PRIVATE Interceptor::SDK)
    if(MSVC)
        target_link_options(InterceptorMinimal PRIVATE "/MANIFESTUAC:level='requireAdministrator' uiAccess='false'")
    endif()
endif()
